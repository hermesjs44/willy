Paso 3: Vincula todo en tu archivo index.html

**************************

Abre tu index.html y pega las siguientes líneas dentro de la etiqueta <head> para conectar el manifiesto y registrar el Service Worker:
html

**************************

<!-- Enlace al archivo Manifest -->
<link rel="manifest" href="manifest.json">

<!-- Configuración de color para la barra del celular -->
<meta name="theme-color" content="#d32f2f">

<!-- Registro del Service Worker -->
<script>
  if ('serviceWorker' in navigator) {
    window.addEventListener('load', () => {
      navigator.serviceWorker.register('sw.js')
        .then(reg => console.log('Service Worker registrado con éxito', reg))
        .catch(err => console.error('Error al registrar el Service Worker', err));
    });
  }
</script>
